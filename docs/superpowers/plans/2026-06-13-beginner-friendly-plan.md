// Simple flow layout for related term chips
private struct FlowLayout: Layout {
    let spacing: CGFloat

    func sizeThatFits(proposal: ProposedViewSize, subviews: Subviews, cache: inout ()) -> CGSize {
        let rows = arrange(proposal.width ?? 0, subviews)
        let height = rows.last?.maxY ?? 0
        return CGSize(width: proposal.width ?? 0, height: height)
    }

    func placeSubviews(in bounds: CGRect, proposal: ProposedViewSize, subviews: Subviews, cache: inout ()) {
        let rows = arrange(bounds.width, subviews)
        for (rowIdx, row) in rows.enumerated() {
            for item in row {
                subviews[item.index].place(
                    at: CGPoint(x: bounds.minX + item.x, y: bounds.minY + rowIdx * (row.first?.size.height ?? 0) + rowIdx * spacing),
                    proposal: .unspecified
                )
            }
        }
    }

    private func arrange(_ width: CGFloat, _ subviews: Subviews) -> [[(index: Int, x: CGFloat, size: CGSize, maxY: CGFloat)]] {
        var rows: [[(index: Int, x: CGFloat, size: CGSize, maxY: CGFloat)]] = [[]]
        var currentX: CGFloat = 0
        for (idx, subview) in subviews.enumerated() {
            let size = subview.sizeThatFits(.unspecified)
            if currentX + size.width > width, !rows[rows.count - 1].isEmpty {
                rows.append([])
                currentX = 0
            }
            rows[rows.count - 1].append((index: idx, x: currentX, size: size, maxY: size.height))
            currentX += size.width + spacing
        }
        return rows
    }
}

import SwiftUI

struct ExplainableTermModifier: ViewModifier {
    let term: String
    @State private var showSheet = false

    func body(content: Content) -> some View {
        if FinanceTerms.shared.lookup(term) != nil {
            Button {
                HapticManager.selection()
                showSheet = true
            } label: {
                content
                    .underline(true, color: Theme.securityGold.opacity(0.5))
            }
            .buttonStyle(.plain)
            .sheet(isPresented: $showSheet) {
                if let entry = FinanceTerms.shared.lookup(term) {
                    TermDefinitionSheet(entry: entry)
                }
            }
        } else {
            content
        }
    }
}

extension View {
    func explainableTerm(_ term: String) -> some View {
        modifier(ExplainableTermModifier(term: term))
    }

    func explainableTermOptional(_ term: String?) -> some View {
        if let term, FinanceTerms.shared.lookup(term) != nil {
            modifier(ExplainableTermModifier(term: term))
        } else {
            self
        }
    }
}
